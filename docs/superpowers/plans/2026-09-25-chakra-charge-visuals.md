# Chakra Charge Visuals Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Holding C shows a Naruto Storm–style charge: a Mixamo "Power Up" pose plus a blue flame aura (two shells, ground ring, flickering light) visible to every player, and charging works even at full chakra.

**Architecture:** `GA_ChakraCharge` drops its full-chakra gate and auto-end, and plays a looping `AM_ChakraCharge` montage (replicated by the ASC). A new cosmetic `UShinobiChakraAuraComponent` on the player character follows the `State.ChakraCharging` tag (already replicated to all machines via `ReplicateActivationOwnedTags`) and fades aura meshes and a light in and out. Materials and the retargeted animation are generated by editor Python scripts.

**Tech Stack:** UE 5.8.2 C++ (GAS), Unreal Python (MaterialEditingLibrary, IKRig/IKRetargeter controllers, `IKRetargetBatchOperation.run_batch_retarget`), Substrate materials (the project has `r.Substrate=True`).

Spec: `docs/superpowers/specs/2026-09-23-combat-slice-design.md` §3 (Chakra row), §5 (Chakra charge aura, Chakra charge pose).

- [ ] **Step 1: Write `Scripts/setup_vfx.py`**

```python
"""Chakra charge aura materials. Idempotent: rebuilds both materials from scratch each run.

M_ChakraAura (flame shells): upward-scrolling procedural noise flames, fresnel rim, soft top fade.
M_ChakraRing (ground ring): a ring pulsing outward plus a faint centre glow.
Both: unlit, additive, two-sided; parameters Color (vector), Intensity, Fade (scalars) driven by
UShinobiChakraAuraComponent.
"""
import unreal

VFX_DIR = "/Game/Shinobi/VFX"
MEL = unreal.MaterialEditingLibrary
TOOLS = unreal.AssetToolsHelpers.get_asset_tools()


def fresh_material(name):
    path = f"{VFX_DIR}/{name}"
    if unreal.EditorAssetLibrary.does_asset_exist(path):
        if not unreal.EditorAssetLibrary.delete_asset(path):
            raise RuntimeError(f"could not delete {path} (is it open in the editor?)")
    material = TOOLS.create_asset(name, VFX_DIR, unreal.Material, unreal.MaterialFactoryNew())
    if material is None:
        raise RuntimeError(f"failed to create {path}")
    material.set_editor_property("blend_mode", unreal.BlendMode.BLEND_ADDITIVE)
    material.set_editor_property("shading_model", unreal.MaterialShadingModel.MSM_UNLIT)
    material.set_editor_property("two_sided", True)
    return material


def node(material, cls, x, y, **props):
    expression = MEL.create_material_expression(material, cls, x, y)
    for key, value in props.items():
        expression.set_editor_property(key, value)
    return expression


def link(src, dst, dst_input, src_output=""):
    if not MEL.connect_material_expressions(src, src_output, dst, dst_input):
        raise RuntimeError(f"failed to connect {src.get_name()}.{src_output} -> {dst.get_name()}.{dst_input}")


def shared_params(material):
    color = node(material, unreal.MaterialExpressionVectorParameter, -1200, -300,
                 parameter_name="Color", default_value=unreal.LinearColor(0.15, 0.55, 1.0, 1.0))
    intensity = node(material, unreal.MaterialExpressionScalarParameter, -1200, -150,
                     parameter_name="Intensity", default_value=6.0)
    fade = node(material, unreal.MaterialExpressionScalarParameter, -1200, 0,
                parameter_name="Fade", default_value=1.0)
    # color * intensity * fade
    ci = node(material, unreal.MaterialExpressionMultiply, -1000, -250)
    link(color, ci, "A")
    link(intensity, ci, "B")
    cif = node(material, unreal.MaterialExpressionMultiply, -850, -200)
    link(ci, cif, "A")
    link(fade, cif, "B")
    return cif


def finish(material, emissive):
    """Substrate: emissive -> Unlit BSDF -> Front Material."""
    unlit = node(material, unreal.MaterialExpressionSubstrateUnlitBSDF, 200, 0)
    link(emissive, unlit, "Emissive Color")
    if not MEL.connect_material_property(unlit, "", unreal.MaterialProperty.MP_FRONT_MATERIAL):
        raise RuntimeError(f"failed to connect Front Material on {material.get_name()}")
    errors = MEL.recompile_material(material)
    if errors:
        raise RuntimeError(f"{material.get_name()} failed to compile: {errors}")
    if not unreal.EditorAssetLibrary.save_loaded_asset(material):
        raise RuntimeError(f"failed to save {material.get_name()}")
    unreal.log(f"[setup_vfx] built: {material.get_path_name()}")


def build_aura():
    m = fresh_material("M_ChakraAura")
    tint = shared_params(m)
    uv = node(m, unreal.MaterialExpressionTextureCoordinate, -1400, 250, u_tiling=4.0, v_tiling=1.5)
    pan = node(m, unreal.MaterialExpressionPanner, -1200, 250, speed_x=0.0, speed_y=-1.2)  # flames rise
    link(uv, pan, "Coordinate")
    time = node(m, unreal.MaterialExpressionTime, -1200, 400)
    slow = node(m, unreal.MaterialExpressionMultiply, -1050, 400, const_b=0.7)
    link(time, slow, "A")
    pos = node(m, unreal.MaterialExpressionAppendVector, -900, 300)
    link(pan, pos, "A")
    link(slow, pos, "B")
    noise = node(m, unreal.MaterialExpressionNoise, -750, 300, scale=3.0, levels=3, output_min=0.0, output_max=1.0)
    link(pos, noise, "Position")
    flame = node(m, unreal.MaterialExpressionPower, -600, 300, const_exponent=2.0)
    link(noise, flame, "Base")
    # Top fade: 1 at the bottom of the shell, 0 at the top. If the engine cylinder's V runs the other way,
    # the manual check (Task 5) flips this by swapping the OneMinus.
    v = node(m, unreal.MaterialExpressionComponentMask, -1200, 550, r=False, g=True, b=False, a=False)
    raw_uv = node(m, unreal.MaterialExpressionTextureCoordinate, -1400, 550)
    link(raw_uv, v, "Input")
    top_fade = node(m, unreal.MaterialExpressionOneMinus, -1050, 550)
    link(v, top_fade, "Input")
    fresnel = node(m, unreal.MaterialExpressionFresnel, -750, 550, exponent=2.5)
    rim = node(m, unreal.MaterialExpressionAdd, -600, 550, const_b=0.35)
    link(fresnel, rim, "A")
    shape = node(m, unreal.MaterialExpressionMultiply, -450, 350)
    link(flame, shape, "A")
    link(top_fade, shape, "B")
    shaped = node(m, unreal.MaterialExpressionMultiply, -300, 400)
    link(shape, shaped, "A")
    link(rim, shaped, "B")
    emissive = node(m, unreal.MaterialExpressionMultiply, -100, 0)
    link(tint, emissive, "A")
    link(shaped, emissive, "B")
    finish(m, emissive)


def build_ring():
    m = fresh_material("M_ChakraRing")
    tint = shared_params(m)
    uv = node(m, unreal.MaterialExpressionTextureCoordinate, -1400, 250)
    centred = node(m, unreal.MaterialExpressionSubtract, -1250, 250, const_b=0.5)
    link(uv, centred, "A")
    dist = node(m, unreal.MaterialExpressionLength, -1100, 250)
    link(centred, dist, "Input")
    radius = node(m, unreal.MaterialExpressionMultiply, -950, 250, const_b=2.0)  # 0 centre .. 1 edge
    link(dist, radius, "A")
    time = node(m, unreal.MaterialExpressionTime, -1250, 450)
    rate = node(m, unreal.MaterialExpressionMultiply, -1100, 450, const_b=1.2)
    link(time, rate, "A")
    pulse = node(m, unreal.MaterialExpressionFrac, -950, 450)  # 0..1 expanding
    link(rate, pulse, "Input")
    gap = node(m, unreal.MaterialExpressionSubtract, -800, 350)
    link(radius, gap, "A")
    link(pulse, gap, "B")
    gap_abs = node(m, unreal.MaterialExpressionAbs, -650, 350)
    link(gap, gap_abs, "Input")
    band = node(m, unreal.MaterialExpressionMultiply, -500, 350, const_b=8.0)  # ring width ~1/8
    link(gap_abs, band, "A")
    ring = node(m, unreal.MaterialExpressionOneMinus, -350, 350)
    link(band, ring, "Input")
    ring_sat = node(m, unreal.MaterialExpressionSaturate, -200, 350)
    link(ring, ring_sat, "Input")
    edge = node(m, unreal.MaterialExpressionOneMinus, -800, 550)  # fade toward the plane edge
    link(radius, edge, "Input")
    edge_sat = node(m, unreal.MaterialExpressionSaturate, -650, 550)
    link(edge, edge_sat, "Input")
    glow = node(m, unreal.MaterialExpressionMultiply, -500, 550, const_b=0.25)
    link(edge_sat, glow, "A")
    body = node(m, unreal.MaterialExpressionAdd, -350, 500)
    link(ring_sat, body, "A")
    link(glow, body, "B")
    shaped = node(m, unreal.MaterialExpressionMultiply, -200, 500)
    link(body, shaped, "A")
    link(edge_sat, shaped, "B")
    emissive = node(m, unreal.MaterialExpressionMultiply, -100, 0)
    link(tint, emissive, "A")
    link(shaped, emissive, "B")
    finish(m, emissive)


build_aura()
build_ring()
```

	UStaticMesh* Cylinder = LoadObject<UStaticMesh>(nullptr, CylinderPath);
	UStaticMesh* Plane = LoadObject<UStaticMesh>(nullptr, PlanePath);
	UMaterialInterface* AuraMaterial = LoadObject<UMaterialInterface>(nullptr, AuraMaterialPath);
	UMaterialInterface* RingMaterial = LoadObject<UMaterialInterface>(nullptr, RingMaterialPath);

	// Engine basic shapes are 100 uu across with the pivot at the centre; this component sits at the
	// capsule centre (192 uu tall), so -94 is just above the floor.
	Pieces.Add(AddPiece(TEXT("AuraInnerShell"), Cylinder, AuraMaterial, FVector(0.f, 0.f, 10.f), FVector(1.1f, 1.1f, 2.2f)));
	Pieces.Add(AddPiece(TEXT("AuraOuterShell"), Cylinder, AuraMaterial, FVector(0.f, 0.f, 25.f), FVector(1.5f, 1.5f, 2.6f)));
	Pieces.Add(AddPiece(TEXT("AuraGroundRing"), Plane, RingMaterial, FVector(0.f, 0.f, -94.f), FVector(3.f, 3.f, 1.f)));

	Light = NewObject<UPointLightComponent>(GetOwner(), TEXT("AuraLight"));
	Light->SetupAttachment(this);
	Light->SetLightColor(Color);
	Light->SetAttenuationRadius(450.f);
	Light->SetCastShadows(false);
	Light->RegisterComponent();

	if (const IAbilitySystemInterface* Owner = Cast<IAbilitySystemInterface>(GetOwner()))
	{
		AbilitySystem = Owner->GetAbilitySystemComponent();
	}
	if (UAbilitySystemComponent* ASC = AbilitySystem.Get())
	{
		TagEventHandle = ASC->RegisterGameplayTagEvent(ShinobiTags::State_ChakraCharging, EGameplayTagEventType::NewOrRemoved)
			.AddUObject(this, &UShinobiChakraAuraComponent::OnChargingTagChanged);
		bCharging = ASC->HasMatchingGameplayTag(ShinobiTags::State_ChakraCharging);
	}
	SetComponentTickEnabled(bCharging);
	UpdatePieces();
}

void UShinobiChakraAuraComponent::EndPlay(const EEndPlayReason::Type EndPlayReason)
{
	if (UAbilitySystemComponent* ASC = AbilitySystem.Get())
	{
		ASC->UnregisterGameplayTagEvent(TagEventHandle, ShinobiTags::State_ChakraCharging, EGameplayTagEventType::NewOrRemoved);
	}
	Super::EndPlay(EndPlayReason);
}

UStaticMeshComponent* UShinobiChakraAuraComponent::AddPiece(FName Name, UStaticMesh* Mesh, UMaterialInterface* Material, const FVector& Offset, const FVector& Scale)
{
	UStaticMeshComponent* Piece = NewObject<UStaticMeshComponent>(GetOwner(), Name);
	Piece->SetupAttachment(this);
	Piece->SetStaticMesh(Mesh);
	Piece->SetRelativeLocation(Offset);
	Piece->SetRelativeScale3D(Scale);
	Piece->SetCollisionEnabled(ECollisionEnabled::NoCollision);
	Piece->SetGenerateOverlapEvents(false);
	Piece->SetCastShadow(false);
	Piece->RegisterComponent();

	UMaterialInstanceDynamic* Dynamic = Material ? UMaterialInstanceDynamic::Create(Material, this) : nullptr;
	if (Dynamic)
	{
		Dynamic->SetVectorParameterValue(ColorParam, Color);
		Piece->SetMaterial(0, Dynamic);
	}
	PieceMaterials.Add(Dynamic);
	return Piece;
}

void UShinobiChakraAuraComponent::OnChargingTagChanged(const FGameplayTag Tag, int32 NewCount)
{
	bCharging = NewCount > 0;
	SetComponentTickEnabled(true); // fade in or out
}

void UShinobiChakraAuraComponent::TickComponent(float DeltaTime, ELevelTick TickType, FActorComponentTickFunction* ThisTickFunction)
{
	Super::TickComponent(DeltaTime, TickType, ThisTickFunction);
	Fade = FMath::FInterpConstantTo(Fade, bCharging ? 1.f : 0.f, DeltaTime, 1.f / FadeTime);
	Age += DeltaTime;
	if (Pieces.Num() >= 2)
	{
		Pieces[0]->AddRelativeRotation(FRotator(0.f, SpinSpeed * DeltaTime, 0.f));
		Pieces[1]->AddRelativeRotation(FRotator(0.f, -SpinSpeed * DeltaTime, 0.f));
	}
	UpdatePieces();
	if (!bCharging && Fade <= 0.f)
	{
		SetComponentTickEnabled(false);
	}
}

void UShinobiChakraAuraComponent::UpdatePieces()
{
	const bool bVisible = Fade > 0.f;
	for (int32 Index = 0; Index < Pieces.Num(); ++Index)
	{
		Pieces[Index]->SetVisibility(bVisible);
		if (UMaterialInstanceDynamic* Dynamic = PieceMaterials[Index])
		{
			Dynamic->SetScalarParameterValue(FadeParam, Fade);
			Dynamic->SetScalarParameterValue(IntensityParam, Intensity);
		}
	}
	if (Light)
	{
		// ponytail: two-sine flicker; swap for a curve asset if art direction wants a specific rhythm
		const float Flicker = 0.8f + 0.2f * FMath::Sin(Age * 23.f) * FMath::Sin(Age * 7.f);
		Light->SetVisibility(bVisible);
		Light->SetIntensity(LightIntensity * Fade * Flicker);
	}
}
